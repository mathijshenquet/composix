# STEP1: CLONE THE CODE
FROM alpine/git as builder_prepare
WORKDIR /home/
ARG GIT_REPO=https://github.com/mickael-kerjean/filestash
ARG GIT_BRANCH=master
RUN git clone --depth 1 --single-branch --branch ${GIT_BRANCH} ${GIT_REPO}

# STEP2: BUILD BACKEND
FROM golang:1.26-trixie AS builder_backend
WORKDIR /home/filestash/
COPY --from=builder_prepare /home/filestash/ .
RUN apt-get update > /dev/null && \
    apt-get install -y curl make > /dev/null 2>&1 && \
    apt-get install -y libjpeg-dev libtiff-dev libpng-dev libwebp-dev libraw-dev libheif-dev libgif-dev libvips-dev > /dev/null 2>&1 && \
    apt-get install -y libavcodec-dev libavdevice-dev libavfilter-dev libavformat-dev libswresample-dev libswscale-dev libavutil-dev > /dev/null 2>&1 && \
    make init && \
    make build && \
    mkdir -p ./dist/data/state/config/ && \
    cp config/config.json ./dist/data/state/config/config.json

# STEP3: BUILD PROD IMAGE
FROM debian:stable-slim
MAINTAINER mickael@kerjean.me
WORKDIR /app/
COPY --from=builder_backend /home/filestash/dist/ .
RUN apt-get update > /dev/null && \
    apt-get -y upgrade > /dev/null && \
    apt-get install -y --no-install-recommends ca-certificates curl ffmpeg libbrotli1 poppler-utils && \
    useradd filestash && \
    chown -R filestash:filestash /app/ && \
    find /app/data/ -type d -exec chmod 770 {} \; && \
    find /app/data/ -type f -exec chmod 760 {} \; && \
    chmod 730 /app/filestash && \
    rm -rf /var/lib/apt/lists/* && \
    rm -rf /tmp/*

USER filestash
CMD ["/app/filestash"]
EXPOSE 8334

FROM . AS src

BUILDER build
IMPORT ${pkgs.bash} ${pkgs.coreutils} ${pkgs.go_1_26} ${pkgs.gnumake} \
    ${pkgs.gcc} ${pkgs.pkg-config} ${pkgs.cacert} ${pkgs.libjpeg} \
    ${pkgs.libtiff} ${pkgs.libpng} ${pkgs.libwebp} ${pkgs.libraw} \
    ${pkgs.libheif} ${pkgs.giflib} ${pkgs.vips} ${pkgs.ffmpeg} \
    ${pkgs.glib} ${pkgs.lcms2} ${pkgs.zlib} ${pkgs.brotli} \
    ${pkgs.brotli.dev} ${pkgs.libjpeg.dev} ${pkgs.libpng.dev} \
    ${pkgs.libraw.dev} \
    ${pkgs.ffmpeg.dev} ${pkgs.glib.dev} ${pkgs.gcc.cc.lib} \
    ${pkgs.pkgsStatic.libjpeg.out} ${pkgs.pkgsStatic.libpng.out} \
    ${pkgs.pkgsStatic.libwebp} ${pkgs.pkgsStatic.libraw.lib} \
    ${pkgs.pkgsStatic.lcms2.out} ${pkgs.pkgsStatic.zlib.out}
ENV GOMODCACHE = $PWD/.gomodcache
COPY ${src}/context/ .
FETCH go mod download && go generate -x ./server/...
RUN CPATH=${pkgs.brotli.dev}/include:${pkgs.libjpeg.dev}/include:${pkgs.libpng.dev}/include:${pkgs.libwebp}/include:${pkgs.libraw.dev}/include:${pkgs.giflib}/include \
    LIBRARY_PATH=${pkgs.giflib}/lib:${pkgs.pkgsStatic.libwebp}/lib:${pkgs.pkgsStatic.libjpeg.out}/lib:${pkgs.pkgsStatic.libpng.out}/lib:${pkgs.pkgsStatic.zlib.out}/lib:${pkgs.pkgsStatic.libraw.lib}/lib:${pkgs.pkgsStatic.lcms2.out}/lib:${pkgs.gcc.cc.lib}/lib:${pkgs.brotli.out}/lib \
    PKG_CONFIG_PATH=${pkgs.ffmpeg.dev}/lib/pkgconfig:${pkgs.glib.dev}/lib/pkgconfig \
    go build --tags fts5 -o dist/filestash cmd/main.go

SERVICE filestash
COPY ${build}/dist/filestash /bin/filestash
COPY ${src}/context/config/config.json /share/filestash-config.json
COPY setup.sh /bin/filestash-setup
LINK ${pkgs.ffmpeg}/bin/ffmpeg /bin/ffmpeg
LINK ${pkgs.ffmpeg}/bin/ffprobe /bin/ffprobe
LINK ${pkgs.poppler-utils}/bin/pdftotext /bin/pdftotext
LINK ${pkgs.poppler-utils}/bin/pdfinfo /bin/pdfinfo
START_PRE filestash-setup
START filestash
ENV FILESTASH_PATH = /var/lib/filestash
ENV FILESTASH_PORT = 18094
PORT http = $FILESTASH_PORT
STATEDIR /var/lib/filestash
CLAIM egress
